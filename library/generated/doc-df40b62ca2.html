#include <STC15F2K60S2.H>

#define  ON   	0
#define  OFF   	1

unsigned char time0_10ms=0;
unsigned char led=0xff;
unsigned char relay_flag=OFF;
unsigned char buzzer_flag=ON;

void Timer0Init(void)		//1毫秒@12.000MHz
{
	AUXR |= 0x80;		//定时器时钟1T模式
	TMOD &= 0xF0;		//设置定时器模式
	TL0 = 0x20;		//设置定时初值
	TH0 = 0xD1;		//设置定时初值
	TF0 = 0;		//清除TF0标志
	TR0 = 1;		//定时器0开始计时
	
	ET0=1;
	EA=1;
}

void led_output(unsigned char value)
{
	P0=value;
	P2=P2&0x1F|0x80;
	P2&=0x1F;
}
void relay_buzzer(unsigned char relay_value,unsigned char buzzer_value)
{
	P0=0X00;//先将之前的P0的所有位清零
	if(relay_value == ON)	  	P0 |= 0x10;//将第五位置1(高电平),打开继电器
	else 						P0 &= ~0x10;//将第五位置0(低电平),关闭继电器
	if(buzzer_value == ON) 		P0 |= 0x40; //将第七位置1(高电平),打开蜂鸣器
	else                        P0 &= ~0x40;//将第七位置0(低电平),关闭蜂鸣器
	P2=P2&0x1F|0xA0;
	P2&=0x1F;
}
void main(void)
{
	P2&=0x1F;//138译码器初始化
	Timer0Init();
	buzzer_flag=OFF;
	relay_flag=ON;
	led=0xff;
	while(1)
	{
		if(time0_10ms>=10)
		{
			//此处为每10ms要执行的内容
			relay_buzzer(relay_flag,buzzer_flag);
			led_output(led);
			time0_10ms=0;
		}                
	}
}
void Timer0Serviece(void) interrupt 1
{
	//此处为中断函数中要进行的内容
	
	time0_10ms++;
}
